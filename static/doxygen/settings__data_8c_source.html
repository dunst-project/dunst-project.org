85TEST test_valid_parser_and_data_per_type(void)
86{
87 for (size_t i = 0; i < G_N_ELEMENTS(allowed_settings); i++) {
88 struct setting curr = allowed_settings[i];
89 switch (curr.type) {
90 case TYPE_STRING:
91 case TYPE_TIME:
92 case TYPE_DOUBLE:
93 case TYPE_LENGTH:
94 case TYPE_COLOR:
95 case TYPE_GRADIENT:
96 case TYPE_INT: ; // no parser and no parser data needed
97 gchar *error1 = g_strdup_printf("Parser of setting %s should be NULL. It's not needed for this type", curr.name);
98 gchar *error2 = g_strdup_printf("Parser data of setting %s should be NULL. It's not needed for this type", curr.name);
99 ASSERTm(error1, !curr.parser);
100 ASSERTm(error2, !curr.parser_data);
101 g_free(error1);
102 g_free(error2);
103 break;
104 case TYPE_CUSTOM: ; // both parser data and parser are needed
105 gchar *error3 = g_strdup_printf("Parser of setting %s should not be NULL. It's needed for this type", curr.name);
106 gchar *error4 = g_strdup_printf("Parser data of setting %s should not be NULL. It's needed for this type", curr.name);
107 ASSERTm(error3, curr.parser);
108 ASSERTm(error4, curr.parser_data);
109 g_free(error3);
110 g_free(error4);
111 break;
112 case TYPE_LIST: ; // only parser data is needed
113 gchar *error5 = g_strdup_printf("Parser of setting %s should be NULL. It's needed not for this type", curr.name);
114 gchar *error6 = g_strdup_printf("Parser data of setting %s should not be NULL. It's needed for this type", curr.name);
115 ASSERTm(error5, !curr.parser);
116 ASSERTm(error6, curr.parser_data);
117 g_free(error5);
118 g_free(error6);
119 break;
120 case TYPE_PATH: ; // only parser data is neede, but when it's a rule none is needed.
121 gchar *error7 = g_strdup_printf("Parser of setting %s should be NULL. It's needed not for this type", curr.name);
122 gchar *error8 = g_strdup_printf("Parser data of setting %s should not be NULL. It's needed for this type", curr.name);
123 bool is_rule = !curr.value; // if it doesn't have a 'value' it's a rule
124 ASSERTm(error7, !curr.parser);
125 ASSERTm(error8, is_rule || curr.parser_data);
126 g_free(error7);
127 g_free(error8);
128 break;
