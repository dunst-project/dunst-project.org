114/*
115 * Glib leaves the option terminator "--" in the argv after parsing in some
116 * cases. This function gets the specified argv element ignoring the first
117 * terminator.
118 *
119 * See https://docs.gtk.org/glib/method.OptionContext.parse.html for details
120 */
121char *get_argv(char *argv[], int index)
122{
123 for (int i = 0; i <= index; i++) {
124 if (strcmp(argv[i], "--") == 0) {
125 return argv[index + 1];
126 }
127 }
128 return argv[index];
129}
130
131/* Count the number of arguments in argv excluding the terminator "--" */
132int count_args(char *argv[], int argc) {
133 for (int i = 0; i < argc; i++) {
134 if (strcmp(argv[i], "--") == 0)
135 return argc - 1;
136 }
137
138 return argc;
139}

141void parse_commandline(int argc, char *argv[])
142{
143 GError *error = NULL;
144 GOptionContext *context;
145
146 context = g_option_context_new("SUMMARY [BODY]");
147 g_option_context_add_main_entries(context, entries, NULL);
148 if (!g_option_context_parse(context, &argc, &argv, &error)){
149 g_printerr("Invalid commandline: %s\n", error->message);
150 exit(1);
151 }
152
153 g_option_context_free(context);
154
155 if (capabilities) {
156 print_capabilities();
157 die(0);
158 }
159
160 if (serverinfo) {
161 print_serverinfo();
162 die(0);
163 }
164
165 if (say_version) {
166 print_version();
167 die(0);
168 }
169
170 if (*appname == '\0') {
171 g_printerr("Provided appname was empty\n");
172 die(1);
173 }
174
175 if (rest != NULL && rest[0] != NULL) {
176 summary = rest[0];
177
178 if (rest[1] != NULL) {
179 body = g_strcompress(rest[1]);
180
181 if (rest[2] != NULL) {
182 g_printerr("Too many arguments!\n");
183 die(1);
184 }
185 }
186 }
187
188 if (summary == NULL && close_id < 1) {
189 g_printerr("I need at least a summary\n");
190 die(1);
191 }
192
193 if (urgency_str) {
194 switch (urgency_str[0]) {
195 case 'l':
196 case 'L':
197 case '0':
198 urgency = NOTIFY_URGENCY_LOW;
199 break;
200 case 'n':
201 case 'N':
202 case '1':
203 urgency = NOTIFY_URGENCY_NORMAL;
204 break;
205 case 'c':
206 case 'C':
207 case '2':
208 urgency = NOTIFY_URGENCY_CRITICAL;
209 break;
210 default:
211 g_printerr("Unknown urgency: %s\n", urgency_str);
212 g_printerr("Assuming normal urgency\n");
213 break;
214 }
215 }
216}

243void add_action(NotifyNotification *n, char *str)
244{
245 char *action = str;
246 char *label = strchr(str, ',');
247
248 if (!label || *(label+1) == '\0') {
249 g_printerr("Malformed action. Expected \"action,label\", got \"%s\"", str);
250 return;
251 }
252
253 *label = '\0';
254 label++;
255
256 notify_notification_add_action(n, action, label, actioned, NULL, NULL);
257}

259void add_hint(NotifyNotification *n, char *str)
260{
261 char *type = str;
262 char *name = strchr(str, ':');
263 if (!name || *(name+1) == '\0') {
264 g_printerr("Malformed hint. Expected \"type:name:value\", got \"%s\"", str);
265 return;
266 }
267 *name = '\0';
268 name++;
269 char *value = strchr(name, ':');
270 if (!value || *(value+1) == '\0') {
271 g_printerr("Malformed hint. Expected \"type:name:value\", got \"%s\"", str);
272 return;
273 }
274 *value = '\0';
275 value++;
276
277 if (strcmp(type, "int") == 0)
278 notify_notification_set_hint_int32(n, name, atoi(value));
279 else if (strcmp(type, "double") == 0)
280 notify_notification_set_hint_double(n, name, atof(value));
281 else if (strcmp(type, "string") == 0)
282 notify_notification_set_hint_string(n, name, value);
283 else if (strcmp(type, "byte") == 0) {
284 gint h_byte = g_ascii_strtoull(value, NULL, 10);
285 if (h_byte < 0 || h_byte > 0xFF)
286 g_printerr("Not a byte: \"%s\"", value);
287 else
288 notify_notification_set_hint_byte(n, name, (guchar) h_byte);
289 } else
290 g_printerr("Malformed hint. Expected a type of int, double, string or byte, got %s\n", type);
291}

293int main(int argc, char *argv[])
294{
295 setlocale(LC_ALL, "");
296 g_set_prgname(argv[0]);
297
298 #if !GLIB_CHECK_VERSION(2,35,0)
299 g_type_init();
300 #endif
301
302 parse_commandline(argc, argv);
303
304 if (!notify_init(appname)) {
305 g_printerr("Unable to initialize libnotify\n");
306 die(1);
307 }
308
309 // NOTE: Needed to inform libnotify of the spec version
310 notify_get_server_info(NULL, NULL, NULL, NULL);
311
312 NotifyNotification *n = notify_notification_new(summary, body, icon);
313 notify_notification_set_timeout(n, timeout);
314 notify_notification_set_urgency(n, urgency);
315
316 if (category != NULL)
317 notify_notification_set_category(n, category);
318
319 if (transient)
320 notify_notification_set_hint(n, "transient", g_variant_new_boolean(TRUE));
321
322 GError *err = NULL;
323 if (raw_icon_path) {
324 GdkPixbuf *raw_icon = gdk_pixbuf_new_from_file(raw_icon_path, &err);
325
326 if(err) {
327 g_printerr("Unable to get raw icon: %s\n", err->message);
328 die(1);
329 }
330
331 notify_notification_set_image_from_pixbuf(n, raw_icon);
332 }
333
334 if (close_id > 0) {
335 put_id(n, close_id);
336 notify_notification_close(n, &err);
337 if (err) {
338 g_printerr("Unable to close notification: %s\n", err->message);
339 die(1);
340 }
341 die(0);
342 }
343
344 if (replace_id > 0) {
345 put_id(n, replace_id);
346 }
347
348 GMainLoop *l = NULL;
349
350 if (wait || action_strs) {
351 l = g_main_loop_new(NULL, false);
352 g_signal_connect(n, "closed", G_CALLBACK(closed), NULL);
353 }
354
355 if (stack_tag != NULL)
356 notify_notification_set_hint(n, "x-dunst-stack-tag", g_variant_new_string(stack_tag));
357
358 if (action_strs) {
359 for (int i = 0; action_strs[i]; i++) {
360 add_action(n, action_strs[i]);
361 }
362 }
363
364 if (hint_strs) {
365 for (int i = 0; hint_strs[i]; i++) {
366 add_hint(n, hint_strs[i]);
367 }
368 }
369
370
371 notify_notification_show(n, &err);
372 if (err) {
373 g_printerr("Unable to send notification: %s\n", err->message);
374 die(1);
375 }
376
377 if (printid) {
378 g_print("%d\n", get_id(n));
379 fflush(stdout);
380 }
381
382 if (wait || action_strs)
383 g_main_loop_run(l);
384
385 g_object_unref(G_OBJECT (n));
386
387 die(0);
388}
