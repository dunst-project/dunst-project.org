213bool rule_apply_special_filters(struct rule *r, const char *name)
214{
215 if (is_deprecated_section(name)) // shouldn't happen, but just in case
216 return false;
217
218 if (strcmp(name, "global") == 0) {
219 // no filters for global section
220 return true;
221 }
222 if (strcmp(name, "urgency_low") == 0) {
223 r->msg_urgency = URG_LOW;
224 return true;
225 }
226 if (strcmp(name, "urgency_normal") == 0) {
227 r->msg_urgency = URG_NORM;
228 return true;
229 }
230 if (strcmp(name, "urgency_critical") == 0) {
231 r->msg_urgency = URG_CRIT;
232 return true;
233 }
234
235 return false;
236}

281static inline bool rule_field_matches_string(const char *value, const char *pattern)
282{
283 // Always match empty pattern
284 if (STR_EMPTY(pattern))
285 return true;
286
287 // Never match null value
288 if (!value)
289 return false;
290
291 // Use GLib wrapper for PCRE
292 if (settings.enable_pcre) {
293 GError *error = NULL;
294
295 // TODO: When caching regex use G_REGEX_OPTIMIZE
296 GRegex *regex = g_regex_new(pattern, 0, 0, &error);
297 if (error) {
298 LOG_W("Invalid PCRE Regex '%s': %s", pattern, error->message);
299 g_error_free(error);
300 return false;
301 }
302
303 bool matched = g_regex_match(regex, value, 0, NULL);
304 g_regex_unref(regex);
305 return matched;
306 }
307
308 if (settings.enable_regex) {
309 regex_t regex;
310
311 // TODO: Compile each regex only once
312 int err = regcomp(&regex, pattern, REG_NEWLINE | REG_EXTENDED | REG_NOSUB);
313 if (err) {
314 size_t err_size = regerror(err, &regex, NULL, 0);
315 char *err_buf = g_malloc(err_size);
316 regerror(err, &regex, err_buf, err_size);
317 LOG_W("%s: \"%s\"", err_buf, pattern);
318 g_free(err_buf);
319 return false;
320 }
321
322 while (true) {
323 if (regexec(&regex, value, 0, NULL, 0))
324 break;
325 regfree(&regex);
326 return true;
327 }
328 regfree(&regex);
329 return false;
330 }
331
332 // Fallback to fnmatch if no regex is enabled
333 return !fnmatch(pattern, value, 0);
334}

336/*
337 * Check whether rule should be applied to n.
338 */
339bool rule_matches_notification(struct rule *r, struct notification *n)
340{
341 return r->enabled
342 && (r->msg_urgency == URG_NONE || r->msg_urgency == n->urgency)
343 && (r->match_dbus_timeout < 0 || (r->match_dbus_timeout == n->dbus_timeout))
344 && (r->match_transient == -1 || (r->match_transient == n->transient))
345 && rule_field_matches_string(n->appname, r->appname)
346 && rule_field_matches_string(n->desktop_entry, r->desktop_entry)
347 && rule_field_matches_string(n->summary, r->summary)
348 && rule_field_matches_string(n->body, r->body)
349 && rule_field_matches_string(n->iconname, r->icon)
350 && rule_field_matches_string(n->category, r->category)
351 && rule_field_matches_string(n->stack_tag, r->stack_tag);
352}
