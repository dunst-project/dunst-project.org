84int get_expected_dimension_height(int layout_count, int height)
85{
86 // assumes height == notification height, see get_small_max_height
87 int separator_height = (layout_count - 1) * settings.separator_height;
88 int total_gap_size = (layout_count - 1) * settings.gap_size;
89 height *= layout_count;
90 int frame_width_total_height;
91 int expected_height;
92 if(settings.gap_size) {
93 frame_width_total_height = layout_count * (2 * settings.frame_width);
94 expected_height = height + frame_width_total_height + total_gap_size;
95 } else {
96 frame_width_total_height = 2 * settings.frame_width;
97 expected_height = separator_height + height + frame_width_total_height;
98 }
99 return expected_height;
100}

102int get_expected_dimension_y_offset(int layout_count)
103{
104 // assumes settings.height == notification height, see get_small_max_height
105 int expected_y = layout_count * settings.height.max;
106 if(settings.gap_size) {
107 expected_y += (layout_count * (2 * settings.frame_width));
108 expected_y += (layout_count * settings.gap_size);
109 } else {
110 expected_y += (2 * settings.frame_width);
111 expected_y += (layout_count * settings.separator_height);
112 }
113 return expected_y;
114}

248TEST test_layout_render_no_gaps(void)
249{
250 struct length original_height = settings.height;
251 bool orginal_gap_size = settings.gap_size;
252 settings.height = get_small_max_height();
253 settings.gap_size = 0;
254
255 int layout_count;
256 GSList *notifications;
257 GSList *layouts;
258 struct dimensions dim;
259 cairo_surface_t *image_surface;
260 int expected_y;
261
262 layout_count = 3;
263 notifications = get_dummy_notifications(layout_count);
264 layouts = get_dummy_layouts(notifications);
265 dim = calculate_dimensions(layouts);
266 image_surface = cairo_image_surface_create(CAIRO_FORMAT_ARGB32, 1, 1);
267
268 enum corner_pos corners = C_TOP | _C_FIRST;
269 for (GSList *iter = layouts; iter; iter = iter->next) {
270 struct colored_layout *cl_this = iter->data;
271 struct colored_layout *cl_next = iter->next ? iter->next->data : NULL;
272
273 if (!cl_next)
274 corners |= C_BOT | _C_LAST;
275 dim = layout_render(image_surface, cl_this, cl_next, dim, corners);
276
277 corners &= ~(C_TOP | _C_FIRST);
278 }
279
280 expected_y = get_expected_dimension_y_offset(layout_count);
281 ASSERT(dim.y == expected_y);
282
283 g_slist_free_full(layouts, free_colored_layout);
284 g_slist_free_full(notifications, free_dummy_notification);
285 cairo_surface_destroy(image_surface);
286 settings.gap_size = orginal_gap_size;
287 settings.height = original_height;
288
289 PASS();
290}
