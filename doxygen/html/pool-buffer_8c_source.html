25static int anonymous_shm_open(void)
26{
27 char name[] = "/dunst-XXXXXX";
28 int retries = 100;
29 gint64 pid = (gint64)getpid() << 32;
30
31 do {
32 gint64 r = time_now() ^ pid ^ retries;
33 for (int i = 0; i < 6; ++i) {
34 int num = (r & 15) + (r & 16) * 2;
35 name[i + sizeof(name) - 7] = 'A' + num;
36 r >>= 5;
37 }
38
39 --retries;
40
41 // shm_open guarantees that O_CLOEXEC is set
42 int fd = shm_open(name, O_RDWR | O_CREAT | O_EXCL, 0600);
43 if (fd >= 0) {
44 shm_unlink(name);
45 return fd;
46 }
47 } while (retries > 0 && errno == EEXIST);
48
49 return -1;
50}

52static int create_shm_file(off_t size)

77static struct pool_buffer *create_buffer(struct wl_shm *shm, struct pool_buffer *buf,
78 int32_t width, int32_t height)
79{
80 const enum wl_shm_format wl_fmt = WL_SHM_FORMAT_ARGB8888;
81 const cairo_format_t cairo_fmt = CAIRO_FORMAT_ARGB32;
82
83 uint32_t stride = cairo_format_stride_for_width(cairo_fmt, width);
84 size_t size = stride * height;
85
86 void *data = NULL;
87 if (size > 0) {
88 int fd = create_shm_file(size);
89 if (fd == -1) {
90 return NULL;
91 }
92
93 data = mmap(NULL, size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
94 if (data == MAP_FAILED) {
95 close(fd);
96 return NULL;
97 }
98
99 struct wl_shm_pool *pool = wl_shm_create_pool(shm, fd, size);
100 buf->buffer = wl_shm_pool_create_buffer(pool, 0, width, height, stride, wl_fmt);
101 wl_buffer_add_listener(buf->buffer, &buffer_listener, buf);
102 wl_shm_pool_destroy(pool);
103
104 close(fd);
105 }
106
107 buf->data = data;
108 buf->size = size;
109 buf->width = width;
110 buf->height = height;
111 buf->surface = cairo_image_surface_create_for_data(data, cairo_fmt, width, height, stride);
112 buf->cairo = cairo_create(buf->surface);
113 buf->pango = pango_cairo_create_context(buf->cairo);
114 return buf;
115}

117void finish_buffer(struct pool_buffer *buffer)
118{
119 if (buffer->buffer)
120 wl_buffer_destroy(buffer->buffer);
121
122 if (buffer->cairo)
123 cairo_destroy(buffer->cairo);
124
125 if (buffer->surface)
126 cairo_surface_destroy(buffer->surface);
127
128 if (buffer->pango)
129 g_object_unref(buffer->pango);
130
131 if (buffer->data)
132 munmap(buffer->data, buffer->size);
133
134 memset(buffer, 0, sizeof(struct pool_buffer));
135}

137struct pool_buffer *get_next_buffer(struct wl_shm *shm, struct pool_buffer pool[static 2],
138 uint32_t width, uint32_t height)
139{
140 struct pool_buffer *buffer = NULL;
141
142 for (size_t i = 0; i < 2; ++i) {
143 if (pool[i].busy)
144 continue;
145 buffer = &pool[i];
146 }
147
148 if (!buffer)
149 return NULL;
150
151 if (buffer->width != width || buffer->height != height)
152 finish_buffer(buffer);
153
154 if (!buffer->buffer) {
155 if (!create_buffer(shm, buffer, width, height))
156 return NULL;
157 }
158
159 return buffer;
160}
