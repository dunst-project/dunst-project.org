88TEST test_notification_replace_single_field(void)
89{

97 ASSERT_EQ(26, substr - str);

133static struct notification *notification_load_icon_with_scaling(int min_icon_size, int max_icon_size)
134{
135 static bool svg_failed = false;
136
137 struct notification *n = notification_create();
138
139 char *path;
140 GVariant *raw_icon = NULL;
141
142 if (!svg_failed) {
143 path = g_strconcat(base, "/data/icons/valid.svg", NULL); // 16x16
144 raw_icon = notification_setup_raw_image(path);
145
146 if (raw_icon == NULL) {
147 svg_failed = true;
148 printf("Failed to load svg icon, using png\n");
149 g_free(path);
150 }
151 }
152
153 if (raw_icon == NULL) {
154 path = g_strconcat(base, "/data/icons/valid.png", NULL); // 16x16
155 raw_icon = notification_setup_raw_image(path);
156 }
157
158 n->min_icon_size = min_icon_size;
159 n->max_icon_size = max_icon_size;
160 notification_icon_replace_data(n, raw_icon);
161 assert(n->icon_id);
162
163 g_variant_unref(raw_icon);
164 g_free(path);
165
166 return n;
167}
