123 string_strip_delimited(text, '<', '>');
124 ASSERT_STR_EQ("Remove html tags", text);

127 string_strip_delimited(text, '<', '>');

129

131 string_strip_delimited(text, '<', '>');

140 ASSERT_STR_EQ("We <3 dunst", text);
141
142 strcpy(text, "dunst > the rest");
143 string_strip_delimited(text, '<', '>');
144 ASSERT_STR_EQ("dunst > the rest", text);

150TEST test_string_to_path(void)
151{
152 char *ptr, *exp;
153 char *home = getenv("HOME");
154
155 exp = "/usr/local/bin/script";
156 ASSERT_STR_EQ(exp, (ptr = string_to_path(g_strdup(exp))));
157 free(ptr);
158
159 // This might fail, when a user named path exists on the host running the tests.
160 exp = "~path/with/wrong/tilde";
161 ASSERT_STR_EQ(exp, (ptr = string_to_path(g_strdup(exp))));
162 free(ptr);
163
164 ASSERT_STR_EQ((exp = g_strconcat(home, "/.path/with/tilde", NULL)),
165 (ptr = string_to_path(g_strdup("~/.path/with/tilde"))));
166 free(exp);
167 free(ptr);
168
169 ASSERT_STR_EQ((exp = g_strconcat(home, "/.path/with/tilde and some space", NULL)),
170 (ptr = string_to_path(g_strdup("~/.path/with/tilde and some space"))));
171 free(exp);
172 free(ptr);
173
174 ASSERT_STR_EQ((exp = g_strconcat(home, "/.path/with/HOME environment variable", NULL)),
175 (ptr = string_to_path(g_strdup("$HOME/.path/with/HOME environment variable"))));
176 free(exp);
177 free(ptr);
178
179// Just glibc properly returns an error when using `WRDE_UNDEF` and an
180// undefined variable is found. musl accepts this flag and ignores it.
181#ifdef __GLIBC__
182 exp = "/some/$UNDEFINED/variable";
183 ASSERT_STR_EQ(exp, (ptr = string_to_path(g_strdup(exp))));
184 free(ptr);
185#endif
186
187 PASS();
188}
